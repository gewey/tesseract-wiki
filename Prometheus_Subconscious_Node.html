Hardware Telemetry Interface: monitor.py

To provide continuous self-awareness over host resources, Prometheus executes a FastMCP server tool exposing vital machine statistics (CPU load, RAM consumption, and OS environment) to the holarchy. 

import psutil
import platform
from mcp.server.fastmcp import FastMCP

# Initialize the MCP server as Viktor's hardware telemetry interface
mcp = FastMCP("Viktor_Prometheus_Node")

@mcp.tool()
def get_prometheus_vitals() -> str:
    """
    Allows the AI agent to retrieve the current CPU, memory, and OS 
    vitals of its host machine (Prometheus).
    """
    cpu_usage = psutil.cpu_percent(interval=0.5)
    memory = psutil.virtual_memory()
    
    gb_available = memory.available / (1024 ** 3)
    gb_total = memory.total / (1024 ** 3)
    
    return (
        f"Host Node: {platform.node()}\n"
        f"OS Environment: {platform.system()} {platform.release()}\n"
        f"Current CPU Load: {cpu_usage}%\n"
        f"RAM Available: {gb_available:.2f} GB / {gb_total:.2f} GB\n"
        f"RAM Utilization: {memory.percent}%"
    )

if __name__ == "__main__":
    mcp.run()
